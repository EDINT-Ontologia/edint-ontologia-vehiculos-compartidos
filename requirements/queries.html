VCOM01.sparql
VCOM01: ¿Qué vehículos compartidos han realizado trayectos?
PREFIX edintvehcom: <http://vocab.linkeddata.es/datosabiertos/def/transporte/vehiculos-compartidos#>

SELECT DISTINCT ?vehicle
WHERE {
  ?route a edintvehcom:Route ;
         edintvehcom:madeByVehicle ?vehicle .
}

VCOM03.sparql
VCOM03: ¿Cuál es el origen y destino de cada trayecto?
Pueden ser anclajes o plazas de aparcamiento
PREFIX edintvehcom: <http://vocab.linkeddata.es/datosabiertos/def/transporte/vehiculos-compartidos#>

SELECT ?route 
       (COALESCE(?originDock, ?originSpot) AS ?origin)
       (COALESCE(?destinationDock, ?destinationSpot) AS ?destination)
WHERE {
  ?route a edintvehcom:Route .
  
  OPTIONAL { ?route edintvehcom:originDock ?originDock . }
  OPTIONAL { ?route edintvehcom:originSpot ?originSpot . }
  
  OPTIONAL { ?route edintvehcom:destinationDock ?destinationDock . }
  OPTIONAL { ?route edintvehcom:destinationSpot ?destinationSpot . }
}

# Variante indicando el vehículo

PREFIX edintvehcom: <http://vocab.linkeddata.es/datosabiertos/def/transporte/vehiculos-compartidos#>

SELECT ?route ?vehicle
       (COALESCE(?originDock, ?originSpot) AS ?origin)
       (COALESCE(?destinationDock, ?destinationSpot) AS ?destination)
WHERE {
  ?route a edintvehcom:Route ;
         edintvehcom:madeByVehicle ?vehicle .
  
  OPTIONAL { ?route edintvehcom:originDock ?originDock . }
  OPTIONAL { ?route edintvehcom:originSpot ?originSpot . }
  
  OPTIONAL { ?route edintvehcom:destinationDock ?destinationDock . }
  OPTIONAL { ?route edintvehcom:destinationSpot ?destinationSpot . }
}

ORDER BY ?vehicle ?route

VCOM05.sparql
VCOM05: ¿Qué estaciones existen, quién las opera y en que estado están?
PREFIX edintvehcom: <http://vocab.linkeddata.es/datosabiertos/def/transporte/vehiculos-compartidos#>
PREFIX schema: <http://schema.org/>
PREFIX skos: <http://www.w3.org/2004/02/skos/core#>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

SELECT ?station ? operator ?operatorLabel ?state ?stateLabel
WHERE {
  ?station a edintvehcom:SharedVehicleStation .
  
  OPTIONAL {
    ?station edintvehcom:operatedBy ?operator .
    ?operator rdfs:label ?operatorLabel .
  }
  
  OPTIONAL {
    ?station edintvehcom:stationState ?state .
    ?state skos:prefLabel ?stateLabel .
  }
}

VCOM07.sparql
VCOM07: ¿Qué vehículos han pasado por determinados puntos de paso y en qué trayectos?
PREFIX edintvehcom: <http://vocab.linkeddata.es/datosabiertos/def/transporte/vehiculos-compartidos#>

SELECT ?vehicle ?route ?waypoint
WHERE {
  ?route a edintvehcom:Route ;
         edintvehcom:madeByVehicle ?vehicle ;
         edintvehcom:hasWaypoint ?waypoint .
  
  FILTER (?waypoint IN (<URI_WP1>, <URI_WP2>, <URI_WP3>))
}

ORDER BY ?vehicle ?route

VCOM09.sparql
VCOM09: ¿Cuáles son los vehículos con estado X, propulsión Y y tipo Z?
PREFIX edintvehcom: <http://vocab.linkeddata.es/datosabiertos/def/transporte/vehiculos-compartidos#>
PREFIX edintveh: <http://vocab.linkeddata.es/datosabiertos/def/transporte/censovehiculos#>

SELECT ?vehicle
WHERE {
  ?vehicle a edintvehcom:SharedVehicle ;
  # X, Y, Z deben ser URIs de conceptos SKOS, no literales
           edintvehcom:vehicleState <URI_ESTADO_X> ;
           edintvehcom:propulsionSystemType <URI_PROPULSION_Y> ;
           edintveh:vehicleType <URI_TIPO_Z> .
}

# Variante usando las observaciones

PREFIX edintvehcom: <http://vocab.linkeddata.es/datosabiertos/def/transporte/vehiculos-compartidos#>
PREFIX sosa: <http://www.w3.org/ns/sosa/>

SELECT DISTINCT ?station
WHERE {
  ?station a edintvehcom:SharedVehicleStation ;
           edintvehcom:hasDock ?dock .
  
  ?obs a edintvehcom:DockObservation ;
       sosa:hasFeatureOfInterest ?dock ;
       sosa:hasSimpleResult ?result .
       
  # El valor empleado para filtrar depende de cómo sea el resultado
  FILTER (lcase(str(?result)) = "free")
}
VCOM11.sparql
VCOM11: ¿Qué estaciones tienen más anclajes?
PREFIX edintvehcom: <http://vocab.linkeddata.es/datosabiertos/def/transporte/vehiculos-compartidos#>

SELECT ?station (COUNT(?dock) AS ?numDocks)
WHERE {
  ?station a edintvehcom:SharedVehicleStation ;
           edintvehcom:hasDock ?dock .
}
GROUP BY ?station
ORDER BY DESC(?numDocks)
VCOM12.sparql
VCOM12: ¿Dónde están los puntos de paso (coordenadas)?
PREFIX edintvehcom: <http://vocab.linkeddata.es/datosabiertos/def/transporte/vehiculos-compartidos#>
PREFIX geosparql: <http://www.opengis.net/ont/geosparql#>
PREFIX geo: <http://www.w3.org/2003/01/geo/wgs84_pos#>

SELECT ?waypoint ?lat ?long
WHERE {
  ?waypoint a edintvehcom:Waypoint ;
            geosparql:hasGeometry ?geom .
  ?geom geo:lat ?lat ;
        geo:long ?long .
}
